\begin{theorem}[radical_mul_eq_radical_inf]\label{thm:radical_mul_eq_radical_inf}
Let $k$ be an arbitrary field, and let $I,J$ be ideals in
$k[x_1,\ldots,x_n]$. Then
\[
\sqrt{IJ} = \sqrt{I \cap J}.
\]
\end{theorem}

\begin{theorem}[span_X_pair_radical_not_mul]\label{thm:span_X_pair_radical_not_mul}
Let $k$ be an arbitrary field. In $k[x,y]$, let
$I=\langle x\rangle$ and $J=\langle x,y\rangle$.
Then $I$ and $J$ are radical ideals, but $IJ$ is not a radical ideal.
\end{theorem}

\begin{theorem}[radical_mul_ne_mul_radicals_example]\label{thm:radical_mul_ne_mul_radicals_example}
Let $k$ be an arbitrary field. In $k[x,y]$, let
$I=\langle x\rangle$ and $J=\langle x,y\rangle$.
Then
\[
\sqrt{IJ} \neq \sqrt{I}\sqrt{J}.
\]
\end{theorem}
