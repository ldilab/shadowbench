\begin{theorem}[GenLoop]
Let $X$ be a topological space, $x \in X$ and $\Omega^{N}(X,x)$ denote the space of \emph{generalized $N$-loops in $X$ based at $x$} i.e.
\[
\Omega^{N}(X,x)
:=
\Bigl\{\, f : I^{N} \to X \ \text{continuous} \ \Bigm|\ 
f(y)=x \text{ for all } y \in \partial I^{N} \Bigr\}
\]
With respect to the usual topology on $ \Omega^{N}(X,x)$ inherited from the compact-open topology on continuous maps, the evaluation map
\[
\mathrm{ev} : \Omega^{N}(X,x) \times I^{N} \longrightarrow X,
\qquad
(f,y) \longmapsto f(y),
\]
is continuous.  
In particular, for each fixed $y \in I^{N}$, the map
\[
\Omega^{N}(X,x) \longrightarrow X,
\qquad
f \longmapsto f(y),
\]
is continuous.
\end{theorem}
