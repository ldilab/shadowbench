\begin{lemma}[Convergence of the Kochen--Stone weights]
Let $(q_n)_{n\geq1}$ be a non-decreasing sequence in $(0,1)$ such that
$q_n\to q$ as $n\to\infty$. For $n\geq2$, define
\[
u_n=\sum_{1\leq i<j\leq n}q_iq_j,
\qquad
v_n=\sum_{i=1}^{n-1}(n-i)q_i,
\qquad
w_n=\frac{u_n}{v_n}.
\]
Then $w_n\to q$ as $n\to\infty$.
\end{lemma}
