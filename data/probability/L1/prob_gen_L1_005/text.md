\begin{theorem}
Let $\mathbf{X}=(X_1,\dots,X_n)'$ be a random vector
with finite mean vector $\mu=E(\mathbf{X})$ and positive definite
covariance matrix
$V=Cov(\mathbf{X})$. Then
\begin{equation}\label{CMul}
\Pr( (\mathbf{X}-\mu)'V^{-1}(\mathbf{X}-\mu) \geq \varepsilon )\leq \frac n \varepsilon
\end{equation}
for all $\varepsilon>0$
\end{theorem}
