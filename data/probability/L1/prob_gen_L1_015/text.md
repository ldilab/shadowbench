\begin{lemma}[A standard-normal CDF comparison]
Let $m>1$ and $k\in\{1,\ldots,m-1\}$. Define
\[
\beta_k=\frac{1}{1+k}\left(1+\frac1k\right)^{2/3},
\qquad
\gamma_{m,k}=\frac{1}{m-k},
\]
and
\[
\theta=\frac{17}{3\,2^{1/3}}-3\,2^{1/3}.
\]
If $\Phi$ is the cumulative distribution function of the standard normal
distribution, then
\[
\Phi\!\left(
\frac{\beta_k\theta+\frac13\gamma_{m,k}}
{\sqrt{\beta_k+\gamma_{m,k}}}
\right)
\leq
\Phi\!\left(
\frac{\beta_k\theta+\frac13}{\sqrt{\beta_k+1}}
\right).
\]
\end{lemma}
