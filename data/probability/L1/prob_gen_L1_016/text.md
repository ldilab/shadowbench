\begin{proposition}[Sharpness of the $m$-dependent constant]
For a finite family $(A_k)_{1\leq k\leq N}$, write
\[
S_N=\sum_{k=1}^{N}\mathbb{P}(A_k).
\]
Call the family $m$-dependent if, whenever
$I,J\subseteq\{1,\ldots,N\}$ and $|i-j|>m$ for every $i\in I$ and
$j\in J$, the sigma-algebras generated by $(A_i)_{i\in I}$ and
$(A_j)_{j\in J}$ are independent. For every integer $m\geq0$, define
\[
c_m^*=\sup\left\{
c>0:
\begin{array}{l}
\mathbb{P}\!\left(\bigcup_{k=1}^{N}A_k\right)
\geq1-\exp(-cS_N)\\
\text{for every }N\geq1,\text{ every probability space, and every}\\
\text{finite }m\text{-dependent family }(A_k)_{1\leq k\leq N}
\end{array}
\right\}.
\]
Then
\[
c_m^*=\frac{1}{m+1}.
\]
\end{proposition}
