\begin{lemma}[Binomial upper-tail monotonicity between mean jumps]
Let $m\geq2$, let $1\leq k\leq m-1$, and let
$p\in(k/m,(k+1)/m]$. If $B(m,p)$ denotes a binomial random variable
with $m$ trials and success probability $p$, then
\[
\mathbb{P}_{X\sim B(m,p)}\!\left(X\geq\mathbb{E}[X]\right)
\geq
\mathbb{P}_{X\sim B(m,k/m)}(X\geq k+1).
\]
\end{lemma}
