\begin{theorem}[gamma_is_smooth_immersion]
Let $\mathbb{T}^2 = \mathbb{S}^1 \times \mathbb{S}^1 \subseteq \mathbb{C}^2$ denote the torus, and let $\alpha$ be any irrational number. The map $\gamma: \mathbb{R} \to \mathbb{T}^2$ given by
\[ \gamma(t) = \left(e^{2\pi i t}, e^{2\pi i \alpha t}\right) \]
is a smooth immersion.
\end{theorem}
