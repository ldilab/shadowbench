\begin{lemma}
For every nonnegative integer $n$,
\[
\sum_{k=0}^{n}\binom{n+k}{2k}\frac{(-4)^k}{2k+1}
=\frac{(-1)^n}{2n+1},
\]
and
\[
\sum_{k=0}^{n}\binom{n+k}{2k}\frac{(-1)^k}{2k+1}
=
\begin{cases}
\dfrac{(-1)^n}{2n+1},&\text{if }3\nmid 2n+1,\\
\dfrac{2(-1)^{n-1}}{2n+1},&\text{if }3\mid 2n+1.
\end{cases}
\]
\end{lemma}
