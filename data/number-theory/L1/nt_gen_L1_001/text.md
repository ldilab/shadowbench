\begin{lemma}
Let $m$ and $r$ be positive integers, and let $b \ge 2$ be an integer. Write the base-$b$ expansion of $m$ as
\[
m=\sum_{i\ge 0} d_i b^i, \qquad 0\le d_i<b,
\]
with all but finitely many $d_i$ equal to zero, and define its base-$b$ digit sum by $s_b(m)=\sum_{i\ge 0}d_i$. If $b^r-1$ divides $m$, then
\[
s_b(m)\ge (b-1)r.
\]
\end{lemma}
