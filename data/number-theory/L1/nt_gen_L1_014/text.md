\begin{theorem}
For integers $b\ge 2$ and $N\ge 0$, let $c_b(N)$ denote the number of nonzero digits in the base-$b$ expansion of $N$. Let $2\le a<b$ be integers such that $a$ divides $b$. There exists a constant $C>0$, depending only on $a$ and $b$, such that, for every sufficiently large integer $n$ and every integer $N\ge 0$ satisfying
\[
a^n\mid N \qquad\text{and}\qquad b\nmid N,
\]
one has
\[
c_b(N)>C\log n.
\]
Moreover, every constant satisfying
\[
0<C<\frac{1}{\log(\log b/\log a)}
\]
is admissible.
\end{theorem}
