\begin{lemma}
Let $p\ge 5$ be a prime. Define the Fermat quotient to base $2$ by
\[
q_p(2)=\frac{2^{p-1}-1}{p},
\]
and define the harmonic numbers by $H_k=\sum_{j=1}^{k}\frac{1}{j}$. Then
\[
q_p(2)^2\equiv
\sum_{k=1}^{p-1}\frac{H_k}{k\cdot 2^k}
-\sum_{k=1}^{p-1}\frac{2^k}{k^2}
\pmod p.
\]
\end{lemma}
