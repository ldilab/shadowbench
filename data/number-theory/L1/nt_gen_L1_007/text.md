\begin{theorem}
Let $p$ be an odd prime, and let $\alpha,\beta\in\mathbb{Z}_p$, where $\mathbb{Z}_p$ denotes the ring of $p$-adic integers. Then
\[
\sum_{k=0}^{(p-1)/2}
\binom{2k}{k}^{2}
\frac{(\alpha-\beta)^{(p-1)/2-k}(-\beta)^k}{16^k}
\equiv
\sum_{k=0}^{(p-1)/2}
\binom{(p-1)/2}{k}^{2}\alpha^{(p-1)/2-k}\beta^k
\pmod{p^2}.
\]
\end{theorem}
