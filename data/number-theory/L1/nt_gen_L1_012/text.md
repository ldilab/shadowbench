\begin{theorem}
Let $B_t$ denote the $t$th Bell number, the number of partitions of a $t$-element set, and let $S(n,k)$ denote the Stirling number of the second kind, the number of partitions of an $n$-element set into exactly $k$ nonempty subsets. For each positive integer $j$, define
\[
P_j(x)=\sum_{r=0}^{j}B_{j-r}\binom{j}{r}x^r.
\]
Then, for all positive integers $n$ and $j$,
\[
B_{n+j}=\sum_{k=1}^{n}P_j(k)S(n,k).
\]
\end{theorem}
