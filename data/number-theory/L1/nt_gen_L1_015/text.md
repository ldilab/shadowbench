\begin{proposition}
For $n\ge 0$, let
\[
C_n=\frac{1}{n+1}\binom{2n}{n}
\]
be the $n$th Catalan number. Let $p>3$ be a prime and let $a$ be a nonnegative integer. Then
\[
\sum_{n=0}^{p^a-1}C_n\equiv
\begin{cases}
1,&\text{if }p\equiv 1\pmod 3,\text{ or if }p\equiv 2\pmod 3\text{ and }a\text{ is even},\\
-2,&\text{if }p\equiv 2\pmod 3\text{ and }a\text{ is odd}
\end{cases}
\pmod p.
\]
\end{proposition}
