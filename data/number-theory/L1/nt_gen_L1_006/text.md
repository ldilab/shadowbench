\begin{theorem}
For a nonnegative integer $N$, let $c_{10}(N)$ denote the number of nonzero digits in its decimal expansion. Let $(e_k)_{k\ge 1}$ be a sequence of nonnegative integers such that $e_1\ge 1$ and
\[
2^{e_k}>10^{e_{k-1}}
\]
for every $k\ge 2$. If $k\ge 1$, $2^{e_k}$ divides $N$, and $10$ does not divide $N$, then
\[
c_{10}(N)\ge k.
\]
\end{theorem}
