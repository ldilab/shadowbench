\begin{theorem}
For integers $b\ge 2$ and $N\ge 0$, let $c_b(N)$ denote the number of nonzero digits in the base-$b$ expansion of $N$. Let $a,b\ge 2$ be integers, and let $d$ be the smallest positive divisor of $a$ for which
\[
\gcd(a/d,b)=1.
\]
Suppose that $\log d/\log b$ is irrational. Then there is a constant $C>0$, depending only on $a$ and $b$, such that
\[
c_b(a^n)>C\log n
\]
for every sufficiently large positive integer $n$.
\end{theorem}
